// World/UnderwaterDome: what no geometry covers while the head is under water.
// The underside quad ends at the far clip plane (a hairline of sky at the horizon) and a sea has no bed past the last
// terrain tile (the sky shows below the horizon), but the skybox takes no fog: it would look like looking through glass.
// This draws a cube centred on the camera (half-extent = the object's scale, ~100 m, inside the far plane) in the current
// scene fog colour, before the skybox and after the opaque geometry, so only pixels without geometry are filled.
// The cone above ~38 degrees elevation is discarded: that is the Snell window, where the sky shows through the surface.
// Unlit, no textures, no depth texture. Written from scratch.
// AWDK templates/underwater: copy as Shaders/UnderwaterDome.shader.
Shader "World/UnderwaterDome"
{
    Properties
    {
        _WindowSin ("Sine of the elevation where the Snell window starts", Range(0.3, 0.9)) = 0.62
    }
    SubShader
    {
        Tags { "Queue"="Geometry+499" "RenderType"="Opaque" "IgnoreProjector"="True" "ForceNoShadowCasting"="True" }

        Pass
        {
            Name "DOME"
            Cull Off
            ZWrite On
            ZTest LEqual

            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma target 3.0
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"

            float _WindowSin;

            struct appdata
            {
                float4 vertex : POSITION;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                float3 dir : TEXCOORD0;
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                // the cube follows the (per-eye) camera: uniform object scale = half-extent in metres
                float3 offset = v.vertex.xyz * unity_ObjectToWorld._m00;
                o.pos = UnityWorldToClipPos(_WorldSpaceCameraPos + offset);
                o.dir = offset;
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                UNITY_SETUP_STEREO_EYE_INDEX_POST_VERTEX(i);
                clip(_WindowSin - normalize(i.dir).y);
                return fixed4(unity_FogColor.rgb, 1);
            }
            ENDCG
        }
    }
}
