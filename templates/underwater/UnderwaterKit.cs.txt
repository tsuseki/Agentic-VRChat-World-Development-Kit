    /// <summary>
    /// Installer of the local underwater effect (UnderwaterFX.cs). A scene only needs its water volumes:
    ///   UnderwaterKit.Install(footprints, surfaceY, bottomY)        builds the scene root "UnderwaterFX" (re-runnable: replaces it)
    ///   UnderwaterKit.Install(footprints, surfaceY, bottomY, true)  adds another water body (other surface height) to it
    /// A footprint is (xMin, xMax, zMin, zMax) in world metres; the volume reaches from the water surface y down to bottomY.
    /// Install creates what it needs on first use (two quad meshes, two materials, the U# program asset) under the
    /// folder of UnderwaterFX.cs or AssetFolder. When it had to create the program asset it returns PREPARED: wait for
    /// the UdonSharp compile, then call Install again.
    /// Every AudioSource of the scene gets a disabled AudioLowPassFilter and joins the muffle list (sources that already
    /// carry their own active or tuned low-pass filter are skipped and reported).
    /// New objects only: nothing existing is touched except those AudioLowPassFilter components.
    /// All entry points are static and return a plain-text report (call them from the Unity MCP execute_code tool).
    /// </summary>
    public static class UnderwaterKit
    {
        public const string RootName = "UnderwaterFX";

        /// <summary>Folder (Assets/...) for the generated Materials/ and Meshes/. Empty = the folder that holds UnderwaterFX.cs.</summary>
        public static string AssetFolder = "";

        /// <summary>
        /// fogEnd multiplier for both presets when the scene fog is Exponential: with the same fogEnd, Exp fog is much
        /// murkier near the eye than ExpSq (transmittance at 2 m of an 8 m fogEnd: 0.25 vs 0.71); x1.6 gives similar mid-range clarity.
        /// </summary>
        public static float ExpFogScale = 1.6f;

        static readonly CultureInfo Inv = CultureInfo.InvariantCulture;

        // ---- assets -------------------------------------------------------------------------------------------------

        static MonoScript FindScript()
        {
            foreach (var guid in AssetDatabase.FindAssets("UnderwaterFX t:MonoScript"))
            {
                var p = AssetDatabase.GUIDToAssetPath(guid);
                if (Path.GetFileName(p) == "UnderwaterFX.cs") return AssetDatabase.LoadAssetAtPath<MonoScript>(p);
            }
            return null;
        }

        static string ScriptDir(MonoScript script)
        {
            return Path.GetDirectoryName(AssetDatabase.GetAssetPath(script)).Replace('\\', '/') + "/";
        }

        static string Dir()
        {
            if (!string.IsNullOrEmpty(AssetFolder)) return AssetFolder.TrimEnd('/') + "/";
            var script = FindScript();
            return script == null ? null : ScriptDir(script);
        }

        [MenuItem("Tools/AWDK/Underwater/Prepare Assets")]
        static void PrepareMenu() { Debug.Log("[Underwater] " + Prepare()); }

        /// <summary>Creates the missing meshes, materials and the U# program asset. Existing assets are kept as they are.</summary>
        public static string Prepare()
        {
            var script = FindScript();
            if (script == null) return "ERROR UnderwaterFX.cs not found in the project";
            var dir = Dir();
            var sb = new StringBuilder("OK folder=" + dir);

            var so = Shader.Find("World/UnderwaterOverlay");
            var ss = Shader.Find("World/UnderwaterSurface");
            var sd = Shader.Find("World/UnderwaterDome");
            if (so == null || ss == null || sd == null)
                return "ERROR shader missing (copy the three .shader files and let Unity import them): overlay=" + (so != null) + " surface=" + (ss != null) + " dome=" + (sd != null);
            var inc = Path.GetDirectoryName(AssetDatabase.GetAssetPath(so)).Replace('\\', '/') + "/UnderwaterRipples.cginc";
            if (!File.Exists(inc)) return "ERROR " + inc + " missing (copy UnderwaterRipples.cginc next to the shaders)";
            if (ShaderUtil.ShaderHasError(so) || ShaderUtil.ShaderHasError(ss) || ShaderUtil.ShaderHasError(sd))
                return "ERROR shader compile errors: overlay=" + ShaderUtil.ShaderHasError(so) + " surface=" + ShaderUtil.ShaderHasError(ss) + " dome=" + ShaderUtil.ShaderHasError(sd) + " (see the shader inspector)";

            if (AssetDatabase.LoadAssetAtPath<Mesh>(dir + "Meshes/UnderwaterOverlayQuad.asset") == null)
            {
                // overlay quad: only bounds matter (the vertex shader writes clip space); huge bounds so it is never frustum culled
                var oq = new Mesh { name = "UnderwaterOverlayQuad" };
                oq.vertices = new[] { new Vector3(-0.5f, -0.5f, 0), new Vector3(-0.5f, 0.5f, 0), new Vector3(0.5f, 0.5f, 0), new Vector3(0.5f, -0.5f, 0) };
                oq.triangles = new[] { 0, 1, 2, 0, 2, 3 };
                oq.bounds = new Bounds(Vector3.zero, new Vector3(4000, 4000, 4000));
                CreateAsset(oq, dir + "Meshes/UnderwaterOverlayQuad.asset");
                sb.Append(" overlayQuad=created");
            }
            if (AssetDatabase.LoadAssetAtPath<Mesh>(dir + "Meshes/UnderwaterSurfaceQuad.asset") == null)
            {
                // surface quad: 1 x 1 in XZ, faces up, clockwise seen from above (= front face up), so Cull Front draws it from below only
                var sq = new Mesh { name = "UnderwaterSurfaceQuad" };
                sq.vertices = new[] { new Vector3(-0.5f, 0, -0.5f), new Vector3(-0.5f, 0, 0.5f), new Vector3(0.5f, 0, 0.5f), new Vector3(0.5f, 0, -0.5f) };
                sq.normals = new[] { Vector3.up, Vector3.up, Vector3.up, Vector3.up };
                sq.uv = new[] { new Vector2(0, 0), new Vector2(0, 1), new Vector2(1, 1), new Vector2(1, 0) };
                sq.triangles = new[] { 0, 1, 2, 0, 2, 3 };
                sq.RecalculateBounds();
                CreateAsset(sq, dir + "Meshes/UnderwaterSurfaceQuad.asset");
                sb.Append(" surfaceQuad=created");
            }
            if (EnsureMat(dir + "Materials/UnderwaterOverlay.mat", so)) sb.Append(" overlayMat=created");
            if (EnsureMat(dir + "Materials/UnderwaterSurface.mat", ss)) sb.Append(" surfaceMat=created");
            if (AssetDatabase.LoadAssetAtPath<Mesh>(dir + "Meshes/UnderwaterDomeCube.asset") == null)
            {
                // dome: a cube of half-extent 1 (the shader centres it on the camera and scales it by the object's scale); huge bounds so it is never culled
                var dc = new Mesh { name = "UnderwaterDomeCube" };
                dc.vertices = new[] { new Vector3(-1, -1, -1), new Vector3(1, -1, -1), new Vector3(1, 1, -1), new Vector3(-1, 1, -1), new Vector3(-1, -1, 1), new Vector3(1, -1, 1), new Vector3(1, 1, 1), new Vector3(-1, 1, 1) };
                dc.triangles = new[] { 0, 2, 1, 0, 3, 2, 4, 5, 6, 4, 6, 7, 0, 1, 5, 0, 5, 4, 3, 7, 6, 3, 6, 2, 0, 4, 7, 0, 7, 3, 1, 2, 6, 1, 6, 5 };
                dc.bounds = new Bounds(Vector3.zero, new Vector3(4000, 4000, 4000));
                CreateAsset(dc, dir + "Meshes/UnderwaterDomeCube.asset");
                sb.Append(" domeMesh=created");
            }
            if (EnsureMat(dir + "Materials/UnderwaterDome.mat", sd)) sb.Append(" domeMat=created");

            if (UdonSharpEditorUtility.GetUdonSharpProgramAsset(typeof(UnderwaterFX)) == null)
            {
                var pa = ScriptableObject.CreateInstance<UdonSharpProgramAsset>();
                pa.sourceCsScript = script;
                AssetDatabase.CreateAsset(pa, ScriptDir(script) + "UnderwaterFX.asset");
                AssetDatabase.SaveAssets();
                UdonSharpProgramAsset.CompileAllCsPrograms(true);
                sb.Append(" programAsset=created (wait for the UdonSharp compile before Install)");
            }
            AssetDatabase.SaveAssets();
            return sb.ToString();
        }

        static void CreateAsset(Object o, string path)
        {
            Directory.CreateDirectory(Path.GetDirectoryName(path));
            AssetDatabase.CreateAsset(o, path);
        }

        static bool EnsureMat(string path, Shader sh)
        {
            var m = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (m != null)
            {
                if (m.shader != sh) { m.shader = sh; EditorUtility.SetDirty(m); }
                return false;
            }
            m = new Material(sh) { name = Path.GetFileNameWithoutExtension(path) };
            CreateAsset(m, path);
            return true;
        }

        // ---- install ------------------------------------------------------------------------------------------------

        /// <summary>
        /// Builds (append = false: replaces) or extends (append = true) the scene root "UnderwaterFX" with one volume and one
        /// underside quad per footprint (xMin, xMax, zMin, zMax). surfaceY = the water surface, bottomY = below the deepest bed.
        /// undersideMargin grows each underside quad by that many metres per side (large seas: cover what is visible from inside).
        /// </summary>
        public static string Install(Vector4[] footprints, float surfaceY, float bottomY, bool append = false, float undersideMargin = 0f)
        {
            var scene = EditorSceneManager.GetActiveScene();
            if (EditorApplication.isPlaying || !scene.IsValid() || string.IsNullOrEmpty(scene.path)) return "ERROR play mode or no saved scene open";
            if (footprints == null || footprints.Length == 0) return "ERROR no footprints";
            if (bottomY >= surfaceY) return "ERROR bottomY must be below surfaceY";
            for (int i = 0; i < footprints.Length; i++)
                if (footprints[i].y <= footprints[i].x || footprints[i].w <= footprints[i].z) return "ERROR footprint " + i + ": want (xMin, xMax, zMin, zMax)";

            bool hadProgram = UdonSharpEditorUtility.GetUdonSharpProgramAsset(typeof(UnderwaterFX)) != null;
            var prep = Prepare();
            if (prep.StartsWith("ERROR")) return prep;
            if (!hadProgram) return "PREPARED " + prep + " | call Install again once the UdonSharp compile has finished";

            var dir = Dir();
            var overlayMesh = AssetDatabase.LoadAssetAtPath<Mesh>(dir + "Meshes/UnderwaterOverlayQuad.asset");
            var surfaceMesh = AssetDatabase.LoadAssetAtPath<Mesh>(dir + "Meshes/UnderwaterSurfaceQuad.asset");
            var overlayMat = AssetDatabase.LoadAssetAtPath<Material>(dir + "Materials/UnderwaterOverlay.mat");
            var surfaceMat = AssetDatabase.LoadAssetAtPath<Material>(dir + "Materials/UnderwaterSurface.mat");
            if (overlayMesh == null || surfaceMesh == null || overlayMat == null || surfaceMat == null) return "ERROR kit assets missing after Prepare: " + prep;

            GameObject root = FindRoot(scene);
            UnderwaterFX fx = null;
            var volumes = new List<Transform>(); var surfaces = new List<GameObject>(); var bed = new List<float>(); var ripple = new List<float>();
            if (append && root != null)
            {
                fx = root.GetComponent<UnderwaterFX>();
                if (fx == null) return "ERROR root " + RootName + " has no UnderwaterFX: call Install without append";
                if (fx.volumes != null) volumes.AddRange(fx.volumes);
                if (fx.surfaces != null) surfaces.AddRange(fx.surfaces);
                if (fx.bedY != null) bed.AddRange(fx.bedY);
                while (bed.Count < volumes.Count) bed.Add(surfaceY - 1f);
                if (fx.volumeRipple != null) ripple.AddRange(fx.volumeRipple);
                while (ripple.Count < volumes.Count) ripple.Add(1f);
                // an underside that was copied to a per-scene material by SetUnderside keeps using it
                if (surfaces.Count > 0 && surfaces[0] != null)
                {
                    var r0 = surfaces[0].GetComponent<MeshRenderer>();
                    if (r0 != null && r0.sharedMaterial != null) surfaceMat = r0.sharedMaterial;
                }
            }
            else
            {
                if (root != null) Object.DestroyImmediate(root);
                root = new GameObject(RootName);
                UnityEngine.SceneManagement.SceneManager.MoveGameObjectToScene(root, scene);
                Child(root, "Volumes"); Child(root, "Surfaces");
                var overlay = Child(root, "Overlay");
                Visual(overlay, overlayMesh, overlayMat);
                overlay.SetActive(false);
            }
            var volRoot = root.transform.Find("Volumes").gameObject;
            var surfRoot = root.transform.Find("Surfaces").gameObject;

            Physics.SyncTransforms();
            int first = volumes.Count;
            for (int i = 0; i < footprints.Length; i++)
            {
                var f = footprints[i];
                int n = first + i;
                float sx = f.y - f.x, sz = f.w - f.z, sy = surfaceY - bottomY;
                // volume marker: centre = box centre, scale = box size, top face = surfaceY
                var v = Child(volRoot, "Volume_" + n);
                v.transform.position = new Vector3((f.x + f.y) * 0.5f, surfaceY - sy * 0.5f, (f.z + f.w) * 0.5f);
                v.transform.localScale = new Vector3(sx, sy, sz);
                volumes.Add(v.transform);
                bed.Add(MedianBed(f, surfaceY));
                ripple.Add(1f);

                // underside: 1 mm under the surface (Cull Front vs the world water's Cull Back: never the same camera side)
                var s = Child(surfRoot, "Surface_" + n);
                s.transform.position = new Vector3((f.x + f.y) * 0.5f, surfaceY - 0.001f, (f.z + f.w) * 0.5f);
                s.transform.localScale = new Vector3(sx + 2f * undersideMargin, 1f, sz + 2f * undersideMargin);
                Visual(s, surfaceMesh, surfaceMat);
                s.SetActive(false);
                surfaces.Add(s);
            }

            // scene audio sources: a disabled low-pass filter each, listed for the script (voices are not scene sources)
            var srcs = new List<AudioSource>(); var filters = new List<AudioLowPassFilter>(); var skipped = new List<string>();
            foreach (var a in Object.FindObjectsOfType<AudioSource>(true))
            {
                if (a.gameObject.scene != scene || a.transform.IsChildOf(root.transform)) continue;
                var lp = a.GetComponent<AudioLowPassFilter>();
                if (lp != null && (lp.enabled || lp.cutoffFrequency < 21999f)) { skipped.Add(a.name); continue; } // the source's own filter
                if (lp == null) lp = a.gameObject.AddComponent<AudioLowPassFilter>();
                lp.enabled = false; lp.cutoffFrequency = 22000f;
                srcs.Add(a); filters.Add(lp);
            }

            bool created = fx == null;
            if (created) fx = root.AddUdonSharpComponent<UnderwaterFX>();
            fx.volumes = volumes.ToArray(); fx.surfaces = surfaces.ToArray(); fx.overlay = root.transform.Find("Overlay").gameObject; fx.dome = EnsureDome(root);
            fx.mutedSources = srcs.ToArray(); fx.mutedFilters = filters.ToArray(); fx.bedY = bed.ToArray(); fx.volumeRipple = ripple.ToArray();
            var fogNote = "";
            if (created)
            {
                fx.preset = 0;
                if (RenderSettings.fogMode == FogMode.Exponential)
                {
                    fx.fogEnd = new[] { 8f * ExpFogScale, 4f * ExpFogScale };
                    fogNote = " (Exponential world: fogEnd x" + ExpFogScale.ToString(Inv) + ")";
                }
            }
            UdonSharpEditorUtility.CopyProxyToUdon(fx);
            UdonSharpEditorUtility.GetBackingUdonBehaviour(fx).SyncMethod = VRC.SDKBase.Networking.SyncType.None;
            EditorUtility.SetDirty(fx);

            EditorSceneManager.MarkSceneDirty(scene);
            var sb = new StringBuilder("OK volumes=" + volumes.Count + " (+" + footprints.Length + ") surfaces=" + surfaces.Count);
            sb.Append(" audioSources=" + srcs.Count + " surfaceY=" + surfaceY.ToString("F3", Inv) + " bottomY=" + bottomY.ToString("F3", Inv));
            sb.Append(" bedY=");
            for (int i = 0; i < bed.Count; i++) sb.Append((i > 0 ? "," : "") + bed[i].ToString("F2", Inv));
            sb.Append(" fogMode=" + RenderSettings.fogMode + " fogEnd=" + fx.fogEnd[0].ToString("F1", Inv) + "/" + fx.fogEnd[1].ToString("F1", Inv) + fogNote);
            if (skipped.Count > 0) sb.Append(" skippedAudio(own low-pass)=" + string.Join(",", skipped));
            if (!RenderSettings.fog) sb.Append(" WARNING: scene fog is OFF; builds then contain no fog variants and the underwater fog will not render: turn fog on (any mode) first");
            sb.Append(" | " + ScanTransparents());
            sb.Append(" | save the scene");
            return sb.ToString();
        }

        [MenuItem("Tools/AWDK/Underwater/Install On Selected Water")]
        static void InstallSelectionMenu() { Debug.Log("[Underwater] " + InstallFromRenderers(Selection.gameObjects)); }

        /// <summary>
        /// One water body per object: footprint = the world bounds of its renderers (axis-aligned), surface = bounds top,
        /// bottom = 0.5 m under the lowest collider found under the footprint. The first replaces the root, the rest append.
        /// Meant for flat water planes; check surfaceY in the report and use Install with measured values otherwise.
        /// </summary>
        public static string InstallFromRenderers(GameObject[] waters)
        {
            if (waters == null || waters.Length == 0) return "ERROR select the water surface objects first";
            Physics.SyncTransforms();
            var sb = new StringBuilder();
            bool append = false;
            foreach (var w in waters)
            {
                var rs = w.GetComponentsInChildren<Renderer>(true);
                if (rs.Length == 0) { sb.Append(w.name + ": no renderer; "); continue; }
                var b = rs[0].bounds;
                for (int i = 1; i < rs.Length; i++) b.Encapsulate(rs[i].bounds);
                var f = new Vector4(b.min.x, b.max.x, b.min.z, b.max.z);
                float surface = b.max.y;
                float low = LowestBed(f, surface);
                float bottom = float.IsNaN(low) ? surface - 10f : low - 0.5f;
                var res = Install(new[] { f }, surface, bottom, append);
                sb.Append(w.name + ": " + res + "; ");
                if (!res.StartsWith("OK")) break;
                append = true;
            }
            return sb.ToString();
        }

        /// <summary>
        /// A rotated rectangular water body (a pool): volumes are axis-aligned boxes, so the rectangle is covered by up to `slabs`
        /// inscribed slabs along x (no volume overhangs the pool: the head is never "under water" beside it), and every slab's underside
        /// quad is the exact rotated rectangle. corners = the four corners in order (p0 -> p1 -> p2 -> p3, x/z in metres).
        /// Appends to the scene's UnderwaterFX. bedY (the caustics plane of its volumes) defaults to the measured median bed.
        /// </summary>
        public static string InstallRect(Vector2[] corners, float surfaceY, float bottomY, int slabs = 4, float bedY = float.NaN)
        {
            if (corners == null || corners.Length != 4) return "ERROR want the 4 corners of the rectangle";
            Vector2 c = (corners[0] + corners[1] + corners[2] + corners[3]) * 0.25f;
            Vector2 u = corners[1] - corners[0], v = corners[3] - corners[0];
            float xmin = float.MaxValue, xmax = float.MinValue;
            foreach (var p in corners) { xmin = Mathf.Min(xmin, p.x); xmax = Mathf.Max(xmax, p.x); }
            var feet = new List<Vector4>();
            float area = 0f, polyArea = u.magnitude * v.magnitude;
            for (int s = 0; s < Mathf.Max(slabs, 1); s++)
            {
                float xa = Mathf.Lerp(xmin, xmax, (float)s / slabs), xb = Mathf.Lerp(xmin, xmax, (float)(s + 1) / slabs);
                float lo = float.MinValue, hi = float.MaxValue; bool ok = true;
                foreach (var x in new[] { xa, xb })
                {
                    float zl, zh;
                    if (!Section(corners, Mathf.Clamp(x, xmin + 0.002f, xmax - 0.002f), out zl, out zh)) { ok = false; break; }
                    lo = Mathf.Max(lo, zl); hi = Mathf.Min(hi, zh);
                }
                if (!ok) continue;
                lo -= RectOverhang; hi += RectOverhang;     // pool rims: a few cm of volume past the water edge does no harm
                if (hi - lo < 0.3f || xb - xa < 0.1f) continue;
                feet.Add(new Vector4(xa, xb, lo, hi)); area += (xb - xa) * (hi - lo);
            }
            if (feet.Count == 0) return "ERROR no slab fits inside the rectangle";
            var res = Install(feet.ToArray(), surfaceY, bottomY, true);
            if (!res.StartsWith("OK")) return res;
            var fx = FindFx();
            float yaw = Mathf.Atan2(-u.y, u.x) * Mathf.Rad2Deg;
            for (int i = fx.surfaces.Length - feet.Count; i < fx.surfaces.Length; i++)
            {
                var t = fx.surfaces[i].transform;
                t.SetPositionAndRotation(new Vector3(c.x, surfaceY - 0.001f, c.y), Quaternion.Euler(0f, yaw, 0f));
                t.localScale = new Vector3(u.magnitude, 1f, v.magnitude);
                if (!float.IsNaN(bedY)) fx.bedY[i] = bedY;
                EditorUtility.SetDirty(t);
            }
            UdonSharpEditorUtility.CopyProxyToUdon(fx);
            EditorUtility.SetDirty(fx);
            return res + " | rect slabs=" + feet.Count + " coverage=" + (100f * area / polyArea).ToString("F0", Inv) + "% yaw=" + yaw.ToString("F1", Inv);
        }

        // z range of the convex polygon at x (false when x is outside it)
        static bool Section(Vector2[] poly, float x, out float zlo, out float zhi)
        {
            zlo = float.MaxValue; zhi = float.MinValue;
            for (int i = 0; i < poly.Length; i++)
            {
                Vector2 a = poly[i], b = poly[(i + 1) % poly.Length];
                if ((a.x - x) * (b.x - x) > 0f || Mathf.Approximately(a.x, b.x)) continue;
                float z = a.y + (b.y - a.y) * (x - a.x) / (b.x - a.x);
                zlo = Mathf.Min(zlo, z); zhi = Mathf.Max(zhi, z);
            }
            return zhi > zlo;
        }

        // ---- per-world look -----------------------------------------------------------------------------------------

        /// <summary>Colours (and optionally the fog distance) of one preset, 0 Subtle / 1 Strong, on this scene's UnderwaterFX.</summary>
        public static string SetLook(int preset, Color tint, Color fogColor, Color murkColor, float fogEnd = -1f)
        {
            var fx = FindFx();
            if (fx == null) return "ERROR no " + RootName + " in the open scene: Install first";
            int p = Mathf.Clamp(preset, 0, 1);
            fx.tint[p] = tint; fx.fogColor[p] = fogColor; fx.murkColor[p] = murkColor;
            if (fogEnd > 0f) fx.fogEnd[p] = fogEnd;
            return Commit(fx, "OK preset " + p + " tint/fog/murk set, fogEnd=" + fx.fogEnd[p].ToString("F1", Inv));
        }

        /// <summary>Ripple multiplier of one water body (volume index as in the Install report): calm pond 0.5, open water 1.5.</summary>
        public static string SetRipple(int volume, float multiplier)
        {
            var fx = FindFx();
            if (fx == null || fx.volumes == null || volume < 0 || volume >= fx.volumes.Length) return "ERROR no " + RootName + " or no volume " + volume;
            var r = new float[fx.volumes.Length];
            for (int i = 0; i < r.Length; i++) r[i] = fx.volumeRipple != null && i < fx.volumeRipple.Length ? fx.volumeRipple[i] : 1f;
            r[volume] = Mathf.Max(0f, multiplier);
            fx.volumeRipple = r;
            return Commit(fx, "OK volume " + volume + " ripple x" + r[volume].ToString("F2", Inv));
        }

        /// <summary>Grows the underside quad of one water body to its volume plus `margin` metres per side (seas: reach what is visible from inside, up to where the world's water ends or the fog is opaque).</summary>
        public static string SetUndersideMargin(int volume, float margin)
        {
            var fx = FindFx();
            if (fx == null || fx.volumes == null || fx.surfaces == null || volume < 0 || volume >= fx.volumes.Length || volume >= fx.surfaces.Length) return "ERROR no " + RootName + " or no volume " + volume;
            var vs = fx.volumes[volume].lossyScale; var s = fx.surfaces[volume].transform;
            s.localScale = new Vector3(vs.x + 2f * margin, 1f, vs.z + 2f * margin);
            EditorUtility.SetDirty(s);
            return Commit(fx, "OK volume " + volume + " underside quad " + s.localScale.x.ToString("F0", Inv) + " x " + s.localScale.z.ToString("F0", Inv) + " m");
        }

        /// <summary>The preset the world ships with: 0 Subtle (default), 1 Strong.</summary>
        public static string SetPreset(int preset)
        {
            var fx = FindFx();
            if (fx == null) return "ERROR no " + RootName + " in the open scene: Install first";
            fx.preset = Mathf.Clamp(preset, 0, 1);
            return Commit(fx, "OK preset=" + fx.preset);
        }

        /// <summary>
        /// Underside colours for this scene only: copies the shared material to Materials/UnderwaterSurface_&lt;Scene&gt;.mat
        /// (once) and assigns it to this scene's Surface_* quads, so other worlds of the project keep theirs.
        /// </summary>
        public static string SetUnderside(Color color, Color mirrorColor)
        {
            var fx = FindFx();
            if (fx == null || fx.surfaces == null || fx.surfaces.Length == 0) return "ERROR no " + RootName + " with surfaces in the open scene";
            var scene = EditorSceneManager.GetActiveScene();
            var path = Dir() + "Materials/UnderwaterSurface_" + scene.name + ".mat";
            var m = AssetDatabase.LoadAssetAtPath<Material>(path);
            if (m == null)
            {
                var shared = AssetDatabase.LoadAssetAtPath<Material>(Dir() + "Materials/UnderwaterSurface.mat");
                if (shared == null) return "ERROR shared UnderwaterSurface.mat missing: run Prepare";
                m = new Material(shared) { name = Path.GetFileNameWithoutExtension(path) };
                CreateAsset(m, path);
            }
            m.SetColor("_Color", color); m.SetColor("_MirrorColor", mirrorColor);
            EditorUtility.SetDirty(m);
            foreach (var s in fx.surfaces)
            {
                if (s == null) continue;
                var r = s.GetComponent<MeshRenderer>();
                if (r != null) { r.sharedMaterial = m; EditorUtility.SetDirty(r); }
            }
            AssetDatabase.SaveAssets();
            EditorSceneManager.MarkSceneDirty(scene);
            return "OK underside material " + path + " on " + fx.surfaces.Length + " surfaces | save the scene";
        }

        static string Commit(UnderwaterFX fx, string msg)
        {
            UdonSharpEditorUtility.CopyProxyToUdon(fx);
            EditorUtility.SetDirty(fx);
            EditorSceneManager.MarkSceneDirty(fx.gameObject.scene);
            return msg + " | save the scene";
        }

        /// <summary>Caustics plane (Strong) of one water body, metres: use it when the median bed of a sea or a pool is not where players see the bed.</summary>
        public static string SetBedY(int volume, float y)
        {
            var fx = FindFx();
            if (fx == null || fx.volumes == null || volume < 0 || volume >= fx.volumes.Length) return "ERROR no " + RootName + " or no volume " + volume;
            var b = new float[fx.volumes.Length];
            for (int i = 0; i < b.Length; i++) b[i] = fx.bedY != null && i < fx.bedY.Length ? fx.bedY[i] : y;
            b[volume] = y;
            fx.bedY = b;
            return Commit(fx, "OK volume " + volume + " bedY=" + y.ToString("F2", Inv));
        }

        /// <summary>
        /// Re-collects the scene's AudioSources into the muffle list without rebuilding anything (run it after ambience, waves or video
        /// were added to a scene that already has its UnderwaterFX). Same rules as Install: a disabled low-pass filter per source;
        /// sources with their own active or tuned low-pass filter are skipped and reported.
        /// </summary>
        public static string RefreshAudio()
        {
            var fx = FindFx();
            if (fx == null) return "ERROR no " + RootName + " in the open scene: Install first";
            var scene = EditorSceneManager.GetActiveScene();
            var root = fx.gameObject;
            var srcs = new List<AudioSource>(); var filters = new List<AudioLowPassFilter>(); var skipped = new List<string>();
            foreach (var a in Object.FindObjectsOfType<AudioSource>(true))
            {
                if (a.gameObject.scene != scene || a.transform.IsChildOf(root.transform)) continue;
                var lp = a.GetComponent<AudioLowPassFilter>();
                bool ours = lp != null && fx.mutedFilters != null && System.Array.IndexOf(fx.mutedFilters, lp) >= 0;
                if (lp != null && !ours && (lp.enabled || lp.cutoffFrequency < 21999f)) { skipped.Add(a.name); continue; }
                if (lp == null) lp = a.gameObject.AddComponent<AudioLowPassFilter>();
                lp.enabled = false; lp.cutoffFrequency = 22000f;
                srcs.Add(a); filters.Add(lp);
            }
            fx.mutedSources = srcs.ToArray(); fx.mutedFilters = filters.ToArray();
            var sb = new StringBuilder("OK audioSources=" + srcs.Count + ":");
            foreach (var s in srcs) sb.Append(" " + s.name);
            if (skipped.Count > 0) sb.Append(" skippedAudio(own low-pass)=" + string.Join(",", skipped));
            return Commit(fx, sb.ToString());
        }

        // the far dome child of the root (created when missing): half-extent DomeSize m around the camera, inactive on land
        public static float DomeSize = 100f;

        static GameObject EnsureDome(GameObject root)
        {
            var t = root.transform.Find("Dome");
            if (t != null) return t.gameObject;
            var dir = Dir();
            var mesh = AssetDatabase.LoadAssetAtPath<Mesh>(dir + "Meshes/UnderwaterDomeCube.asset");
            var mat = AssetDatabase.LoadAssetAtPath<Material>(dir + "Materials/UnderwaterDome.mat");
            if (mesh == null || mat == null) return null;
            var dome = Child(root, "Dome");
            Visual(dome, mesh, mat);
            dome.transform.localScale = Vector3.one * DomeSize;
            dome.SetActive(false);
            return dome;
        }

        /// <summary>Sets the per-preset splash ring look of the scene's UnderwaterFX (ringGain / ringGlow, index 0 = Subtle, 1 = Strong) and re-syncs the Udon program.</summary>
        public static string SetRings(float gainSubtle, float gainStrong, float glowSubtle, float glowStrong)
        {
            var fx = FindFx();
            if (fx == null) return "ERROR no " + RootName + " in the open scene: Install first";
            fx.ringGain = new[] { gainSubtle, gainStrong };
            fx.ringGlow = new[] { glowSubtle, glowStrong };
            return Commit(fx, "OK rings gain " + gainSubtle.ToString("F2", Inv) + "/" + gainStrong.ToString("F2", Inv) + " glow " + glowSubtle.ToString("F2", Inv) + "/" + glowStrong.ToString("F2", Inv));
        }

        /// <summary>Lists what the underside (queue Transparent+50) cannot cover: renderers with a render queue above 3050 (they draw after it, unfogged, and show crisp through the water view), and the world-space canvases that are covered (queue 3000, UI/Default has no fog keywords).</summary>
        public static string ScanTransparents()
        {
            var late = new List<string>(); int canvases = 0;
            var own = FindRootTransform();
            foreach (var r in UnityEngine.Object.FindObjectsOfType<Renderer>())
            {
                if (r.gameObject.scene.name == null || (own != null && r.transform.IsChildOf(own))) continue;
                foreach (var m in r.sharedMaterials)
                    if (m != null && m.renderQueue > 3050) { late.Add(r.transform.name + "(" + m.shader.name + " q" + m.renderQueue + ")"); break; }
            }
            foreach (var c in UnityEngine.Object.FindObjectsOfType<Canvas>())
                if (c.renderMode == RenderMode.WorldSpace && c.isRootCanvas) canvases++;
            return "worldSpaceCanvases=" + canvases + " (covered by the underside) renderersAfterUnderside=" + late.Count + (late.Count > 0 ? ": " + string.Join(", ", late.ToArray(), 0, Mathf.Min(late.Count, 12)) : "");
        }

        static Transform FindRootTransform()
        {
            var r = FindRoot(UnityEngine.SceneManagement.SceneManager.GetActiveScene());
            return r != null ? r.transform : null;
        }

        /// <summary>Adds the far dome (and the dome assets) to a scene whose UnderwaterFX was installed before the dome existed, and re-syncs the component's new fields.</summary>
        public static string AddDome()
        {
            var fx = FindFx();
            if (fx == null) return "ERROR no " + RootName + " in the open scene: Install first";
            var prep = Prepare();
            if (prep.StartsWith("ERROR")) return prep;
            fx.dome = EnsureDome(fx.gameObject);
            if (fx.dome == null) return "ERROR dome assets missing: " + prep;
            if (fx.ringGain == null || fx.ringGain.Length < 2) fx.ringGain = new[] { 1f, 2f };
            if (fx.ringGlow == null || fx.ringGlow.Length < 2) fx.ringGlow = new[] { 0f, 1f };
            return Commit(fx, "OK dome " + fx.dome.name + " scale " + fx.dome.transform.localScale.x.ToString("F0", Inv) + " ringGain=" + fx.ringGain[0].ToString("F1", Inv) + "/" + fx.ringGain[1].ToString("F1", Inv));
        }

        static GameObject FindRoot(UnityEngine.SceneManagement.Scene scene)
        {
            foreach (var r in scene.GetRootGameObjects()) if (r.name == RootName) return r;
            return null;
        }

        static UnderwaterFX FindFx()
        {
            var root = FindRoot(EditorSceneManager.GetActiveScene());
            return root == null ? null : root.GetComponent<UnderwaterFX>();
        }

        // first collider hit straight down from above the surface; NaN when nothing is under the point
        static float Ground(float x, float z, float fromY)
        {
            RaycastHit h;
            return Physics.Raycast(new Vector3(x, fromY, z), Vector3.down, out h, 500f, ~0, QueryTriggerInteraction.Ignore) ? h.point.y : float.NaN;
        }

        // grid step of the bed measurements: 1 m, coarser on seas so a call stays under ~40k rays
        static float GridStep(Vector4 f) { return Mathf.Max(1f, Mathf.Max(f.y - f.x, f.w - f.z) / 200f); }

        /// <summary>Wet cells deeper than this (m under the surface) are ignored by the caustics' bed plane: on a sea the plane should sit at the beds players walk on, not the deep.</summary>
        public static float CausticsMaxDepth = 6f;

        /// <summary>InstallRect: how far (m) a slab volume may reach past the rectangle's edge in z (pool rims).</summary>
        public static float RectOverhang = 0.1f;

        // median bed height over the grid cells of the footprint whose floor is under the water (the caustics' plane)
        static float MedianBed(Vector4 f, float surfaceY)
        {
            var ys = new List<float>(); var shallow = new List<float>();
            float st = GridStep(f);
            for (float x = f.x + st * 0.5f; x < f.y; x += st)
                for (float z = f.z + st * 0.5f; z < f.w; z += st)
                {
                    float y = Ground(x, z, surfaceY + 30f);
                    if (float.IsNaN(y) || y >= surfaceY - 0.05f) continue;
                    ys.Add(y);
                    if (y > surfaceY - CausticsMaxDepth) shallow.Add(y);
                }
            if (shallow.Count > 0) ys = shallow;
            if (ys.Count == 0) return surfaceY - 1f;
            ys.Sort();
            return ys[ys.Count / 2];
        }

        static float LowestBed(Vector4 f, float surfaceY)
        {
            float low = float.NaN;
            float st = GridStep(f);
            for (float x = f.x + st * 0.5f; x < f.y; x += st)
                for (float z = f.z + st * 0.5f; z < f.w; z += st)
                {
                    float y = Ground(x, z, surfaceY + 30f);
                    if (!float.IsNaN(y) && y < surfaceY - 0.05f && (float.IsNaN(low) || y < low)) low = y;
                }
            return low;
        }

        static GameObject Child(GameObject parent, string name)
        {
            var g = new GameObject(name); g.transform.SetParent(parent.transform, false); return g;
        }

        static void Visual(GameObject g, Mesh mesh, Material mat)
        {
            g.AddComponent<MeshFilter>().sharedMesh = mesh;
            var r = g.AddComponent<MeshRenderer>();
            r.sharedMaterial = mat;
            r.shadowCastingMode = ShadowCastingMode.Off; r.receiveShadows = false;
            r.lightProbeUsage = LightProbeUsage.Off; r.reflectionProbeUsage = ReflectionProbeUsage.Off;
            r.motionVectorGenerationMode = MotionVectorGenerationMode.ForceNoMotion;
            r.allowOcclusionWhenDynamic = false;
            GameObjectUtility.SetStaticEditorFlags(g, 0);
        }
    }
