// World/UnderwaterSurface: the water surface seen from BELOW (a world's water surface is usually back-face culled, so
// from under it nothing is there and you would see straight up at the sky).
// Draws only the back faces of an upward-facing quad (Cull Front): invisible from above, so it never fights the
// real water on top. Unlit, alpha blended, fogged by the scene fog (which UnderwaterFX swaps while submerged):
// a rippled bright underside, see-through straight up (the Snell window, approximated) and mirror-bright towards
// grazing angles. The ripples (swell, fine chop, splash rings from players entering the water) come from
// UnderwaterRipples.cginc and its _UdonUW_* globals, so the overlay bends the view with the same surface.
// No reflections, no textures, no depth. Written from scratch.
// AWDK templates/underwater: copy as Shaders/UnderwaterSurface.shader.
Shader "World/UnderwaterSurface"
{
    Properties
    {
        _Color ("Underside colour", Color) = (0.42, 0.78, 0.76, 1)
        _MirrorColor ("Grazing (total reflection) colour", Color) = (0.16, 0.42, 0.42, 1)
        _WindowAlpha ("Alpha looking straight up", Range(0, 1)) = 0.5
        _RippleGlow ("Ripple highlight", Range(0, 1)) = 0.7
    }
    SubShader
    {
        Tags { "Queue"="Transparent-10" "RenderType"="Transparent" "IgnoreProjector"="True" "ForceNoShadowCasting"="True" }

        Pass
        {
            Name "UNDERSIDE"
            Cull Front
            ZWrite Off
            ZTest LEqual
            Blend SrcAlpha OneMinusSrcAlpha

            CGPROGRAM
            #pragma vertex vert
            #pragma fragment frag
            #pragma target 3.0
            #pragma multi_compile_fog
            #pragma multi_compile_instancing
            #include "UnityCG.cginc"
            #include "UnderwaterRipples.cginc"

            fixed4 _Color;
            fixed4 _MirrorColor;
            float _WindowAlpha;
            float _RippleGlow;
            float _UdonUW_RingGlow;

            struct appdata
            {
                float4 vertex : POSITION;
                UNITY_VERTEX_INPUT_INSTANCE_ID
            };

            struct v2f
            {
                float4 pos : SV_POSITION;
                float3 worldPos : TEXCOORD0;
                UNITY_FOG_COORDS(1)
                UNITY_VERTEX_OUTPUT_STEREO
            };

            v2f vert(appdata v)
            {
                v2f o;
                UNITY_SETUP_INSTANCE_ID(v);
                UNITY_INITIALIZE_OUTPUT(v2f, o);
                UNITY_INITIALIZE_VERTEX_OUTPUT_STEREO(o);
                o.pos = UnityObjectToClipPos(v.vertex);
                o.worldPos = mul(unity_ObjectToWorld, v.vertex).xyz;
                UNITY_TRANSFER_FOG(o, o.pos);
                return o;
            }

            fixed4 frag(v2f i) : SV_Target
            {
                UNITY_SETUP_STEREO_EYE_INDEX_POST_VERTEX(i);

                float chopFade = UW_ChopFade(i.worldPos.xz);
                float2 rings;
                float2 slope = UW_SurfaceSlopeRings(i.worldPos.xz, _Time.y, chopFade, rings);

                float3 view = normalize(UnityWorldSpaceViewDir(i.worldPos));
                // surface normal as seen from below (points down), tilted by the ripples
                float3 n = normalize(float3(slope.x, -1.0, slope.y));
                float ndv = saturate(dot(n, view));

                // window straight up -> total reflection beyond ~48 degrees from vertical (cos 0.66)
                float mirror = 1.0 - smoothstep(0.58, 0.74, ndv);
                float fres = pow(1.0 - ndv, 3.0);
                float alpha = saturate(max(lerp(_WindowAlpha, 0.97, fres), mirror * 0.97));

                // ripple glints: slope magnitude lights the crests (18 = the old 2.2 on the slope before its 0.35 tilt)
                float glint = saturate(dot(slope, slope) * 18.0);
                float3 col = lerp(_Color.rgb, _MirrorColor.rgb, mirror);
                col += _RippleGlow * glint * _Color.rgb;

                // splash ring highlight (a preset whose mirror band hides the rings sets _UdonUW_RingGlow > 0): rings read as bright as the swell's glints
                float ringGlint = saturate(dot(rings, rings) * 40.0) * _UdonUW_RingGlow;
                col += ringGlint * _Color.rgb * 1.2;
                alpha = max(alpha, ringGlint * 0.9);

                fixed4 c = fixed4(col, alpha);
                UNITY_APPLY_FOG(i.fogCoord, c);
                return c;
            }
            ENDCG
        }
    }
}
