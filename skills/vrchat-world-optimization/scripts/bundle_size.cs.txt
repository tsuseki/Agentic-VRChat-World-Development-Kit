// bundle_size: measure a world's upload size without the SDK (Unity MCP execute_code body, edit mode, scene saved).
// Builds the saved scene into one LZ4 (ChunkBased) AssetBundle for the active build target, the way the SDK does.
// The build blocks the editor for ~35-90 s, so it is queued with ONE EditorApplication.delayCall: the call returns at
// once, and the result lands in <outDir>/result.txt ("OK <MB> MB <s>s" or "ERR ..."). Poll for that file from the
// shell (it appears after the build); never re-send while it runs. Edit the three marked values first.
// Reading: raw MB; the SDK .vrcw has come out at ~0.85x of it (two worlds: 168.2/197.6 and 142.4/164.8 MB). Compare
// runs only with each other (A/B one lever at a time; there is no per-asset report). Per-asset-class bundles hung for
// 9+ minutes on a "Moving file failed" modal: build the whole scene only.
string scene = "Assets/MyWorld/MyWorld.unity";   // <-- edit: saved scene path
string outDir = "C:/Temp/bundle_A";              // <-- edit: scratch folder outside the project
string tag = "A";                                // <-- edit: run label
var target = EditorUserBuildSettings.activeBuildTarget;
// Stale VRCSceneDescriptor.DynamicMaterials (left by the last SDK build) drag replaced materials and their textures into
// the bundle; an A/B after a material swap then reads too high (one world: 245.2 MB stale vs 218.6 MB refreshed). The SDK
// refreshes the list itself at build time with VRCWorldAssetExporter.FindDynamicContent, so do the same first (reflection:
// the type is internal to the SDK editor DLL), then save, because the bundle builds the saved scene.
string refresh = "FindDynamicContent not found";
foreach (var asm in System.AppDomain.CurrentDomain.GetAssemblies()) {
    System.Type[] types; try { types = asm.GetTypes(); } catch { continue; }
    foreach (var ty in types) {
        if (ty.Name != "VRCWorldAssetExporter") continue;
        var m = ty.GetMethod("FindDynamicContent", System.Reflection.BindingFlags.Static | System.Reflection.BindingFlags.Public | System.Reflection.BindingFlags.NonPublic);
        if (m == null) continue;
        var ps = m.GetParameters();
        var desc = Object.FindObjectOfType<VRC.SDK3.Components.VRCSceneDescriptor>();
        if (ps.Length == 0) m.Invoke(null, null);
        else if (ps.Length == 1 && desc != null) m.Invoke(null, new object[] { desc });
        else { refresh = "FindDynamicContent: unexpected signature (" + ps.Length + " params)"; break; }
        UnityEditor.SceneManagement.EditorSceneManager.MarkSceneDirty(UnityEngine.SceneManagement.SceneManager.GetActiveScene());
        UnityEditor.SceneManagement.EditorSceneManager.SaveOpenScenes();
        refresh = "DynamicMaterials refreshed (" + (desc != null ? desc.DynamicMaterials.Count.ToString() : "?") + ")";
        break;
    }
}
System.IO.Directory.CreateDirectory(outDir);
EditorApplication.delayCall += () => {
    var t0 = System.DateTime.Now; string res;
    try {
        var b = new AssetBundleBuild { assetBundleName = tag + ".vrcw", assetNames = new[] { scene } };
        BuildPipeline.BuildAssetBundles(outDir, new[] { b }, BuildAssetBundleOptions.ChunkBasedCompression | BuildAssetBundleOptions.ForceRebuildAssetBundle, target);
        var fi = new System.IO.FileInfo(System.IO.Path.Combine(outDir, tag + ".vrcw"));
        res = "OK " + (fi.Exists ? (fi.Length / 1048576.0).ToString("0.0", System.Globalization.CultureInfo.InvariantCulture) + " MB" : "missing") + " " + (System.DateTime.Now - t0).TotalSeconds.ToString("0") + "s";
    } catch (System.Exception e) { res = "ERR " + e.Message; }
    System.IO.File.WriteAllText(outDir + "/result.txt", res);
};
return "queued target=" + target + "; " + refresh;
